from jwcrypto import jwk
import uuid
from starlette.responses import JSONResponse
from starlette.requests import Request
from typing import Any

import jwt
import time
import json
import hashlib
import httpx
import logging

from jwt import PyJWK, PyJWKClient

logger = logging.getLogger(__name__)
AUTH_HEADER_PREFIX = "Bearer "


class PushNotificationAuth:
    def _calculate_request_body_sha256(self, data: dict[str, Any]):
        """Calculates the SHA256 hash of a request body.

        This logic needs to be same for both the agent who signs the payload and the client verifier.
        """
        body_str = json.dumps(
            data,
            ensure_ascii=False,
            allow_nan=False,
            indent=None,
            separators=(",", ":"),
        )
        return hashlib.sha256(body_str.encode()).hexdigest()


class PushNotificationSenderAuth(PushNotificationAuth):
    def __init__(self):
        self.public_keys = []
        self.private_key_jwk: PyJWK = None

    @staticmethod
    async def verify_push_notification_url(url: str) -> bool:
        async with httpx.AsyncClient(timeout=10) as client:
            try:
                validation_token = str(uuid.uuid4())
                response = await client.get(
                    url, params={"validationToken": validation_token}
                )
                response.raise_for_status()
                is_verified = response.text == validation_token

                logger.info(f"Verified push-notification URL: {url} => {is_verified}")
                return is_verified
            except Exception as e:
                logger.warning(
                    f"Error during sending push-notification for URL {url}: {e}"
                )

        return False

    def generate_jwk(self):
        key = jwk.JWK.generate(kty="RSA", size=2048, kid=str(uuid.uuid4()), use="sig")
        self.public_keys.append(key.export_public(as_dict=True))
        self.private_key_jwk = PyJWK.from_json(key.export_private())

    def handle_jwks_endpoint(self, _request: Request):
        """Allow clients to fetch public keys."""
        return JSONResponse({"keys": self.public_keys})

    def _generate_jwt(self, data: dict[str, Any]):
        """JWT is generated by signing both the request payload SHA digest and time of token generation.

        Payload is signed with private key and it ensures the integrity of payload for client.
        Including iat prevents from replay attack.
        """

        iat = int(time.time())

        return jwt.encode(
            {
                "iat": iat,
                "request_body_sha256": self._calculate_request_body_sha256(data),
            },
            key=self.private_key_jwk,
            headers={"kid": self.private_key_jwk.key_id},
            algorithm="RS256",
        )

    async def send_push_notification(self, url: str, data: dict[str, Any]):
        jwt_token = self._generate_jwt(data)
        headers = {"Authorization": f"Bearer {jwt_token}"}
        async with httpx.AsyncClient(timeout=10) as client:
            try:
                response = await client.post(url, json=data, headers=headers)
                response.raise_for_status()
                logger.info(f"Push-notification sent for URL: {url}")
            except Exception as e:
                logger.warning(
                    f"Error during sending push-notification for URL {url}: {e}"
                )


class PushNotificationReceiverAuth(PushNotificationAuth):
    def __init__(self):
        self.public_keys_jwks = []
        self.jwks_client = None

    async def load_jwks(self, jwks_url: str):
        self.jwks_client = PyJWKClient(jwks_url)

    async def verify_push_notification(self, request: Request) -> bool:
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith(AUTH_HEADER_PREFIX):
            print("Invalid authorization header")
            return False

        token = auth_header[len(AUTH_HEADER_PREFIX) :]
        signing_key = self.jwks_client.get_signing_key_from_jwt(token)

        decode_token = jwt.decode(
            token,
            signing_key,
            options={"require": ["iat", "request_body_sha256"]},
            algorithms=["RS256"],
        )

        actual_body_sha256 = self._calculate_request_body_sha256(await request.json())
        if actual_body_sha256 != decode_token["request_body_sha256"]:
            # Payload signature does not match the digest in signed token.
            raise ValueError("Invalid request body")

        if time.time() - decode_token["iat"] > 60 * 5:
            # Do not allow push-notifications older than 5 minutes.
            # This is to prevent replay attack.
            raise ValueError("Token is expired")

        return True

import json
import time
import hashlib
import uuid
import logging
import httpx
import jwt
from jwt import PyJWK, PyJWKClient, PyJWTError
from starlette.requests import Request
from starlette.responses import JSONResponse
from jwcrypto import jwk
from typing import Any

logger = logging.getLogger(__name__)
AUTH_HEADER_PREFIX = "Bearer "
MAX_TOKEN_AGE_SECONDS = 60 * 5  # 5분


class PushNotificationAuth:
    def _calculate_request_body_sha256(self, data: dict[str, Any]) -> str:
        """Calculates the SHA256 hash of a request body.

        This logic needs to be same for both the agent who signs the payload and the client verifier.
        """
        body_str = json.dumps(
            data,
            ensure_ascii=False,
            allow_nan=False,
            indent=None,
            separators=(",", ":"),
        )
        return hashlib.sha256(body_str.encode()).hexdigest()


class PushNotificationSenderAuth(PushNotificationAuth):
    def __init__(self):
        self.public_keys = []
        self.private_key_jwk: PyJWK | None = None

    @staticmethod
    async def verify_push_notification_url(url: str) -> bool:
        async with httpx.AsyncClient(timeout=10) as client:
            try:
                validation_token = str(uuid.uuid4())
                response = await client.get(
                    url, params={"validationToken": validation_token}
                )
                response.raise_for_status()
                is_verified = response.text == validation_token

                # [방어 강화] URL 원본 로깅을 통한 정보 노출 방지 (마스킹 또는 결과 여부만 기록)
                logger.info(f"Verified push-notification URL endpoint status => {is_verified}")
                return is_verified
            except httpx.HTTPStatusError as e:
                logger.warning(f"HTTP error during push-notification URL verification: Status {e.response.status_code}")
            except httpx.RequestError as e:
                logger.warning(f"Network error during push-notification URL verification: {type(e).__name__}")
            except Exception as e:
                logger.warning(f"Unexpected error during push-notification URL verification: {type(e).__name__}")

        return False

    def generate_jwk(self):
        key = jwk.JWK.generate(kty="RSA", size=2048, kid=str(uuid.uuid4()), use="sig")
        self.public_keys.append(key.export_public(as_dict=True))
        self.private_key_jwk = PyJWK.from_json(key.export_private())

    def handle_jwks_endpoint(self, _request: Request):
        """Allow clients to fetch public keys."""
        return JSONResponse({"keys": self.public_keys})

    def _generate_jwt(self, data: dict[str, Any]) -> str:
        if not self.private_key_jwk:
            raise RuntimeError("Private key JWK is not generated. Call generate_jwk() first.")

        iat = int(time.time())
        return jwt.encode(
            {
                "iat": iat,
                "request_body_sha256": self._calculate_request_body_sha256(data),
            },
            key=self.private_key_jwk.key.key_id, # 수정된 key key_id 호환성 준수
            headers={"kid": self.private_key_jwk.key_id},
            algorithm="RS256",
        )

    async def send_push_notification(self, url: str, data: dict[str, Any]) -> bool:
        try:
            jwt_token = self._generate_jwt(data)
            headers = {"Authorization": f"Bearer {jwt_token}"}
            async with httpx.AsyncClient(timeout=10) as client:
                response = await client.post(url, json=data, headers=headers)
                response.raise_for_status()
                logger.info("Push-notification successfully transmitted.")
                return True
        except httpx.HTTPStatusError as e:
            logger.warning(f"Push notification failed with HTTP status: {e.response.status_code}")
        except httpx.RequestError as e:
            logger.warning(f"Push notification network failure: {type(e).__name__}")
        except Exception as e:
            logger.warning(f"Unexpected error while sending push notification: {type(e).__name__}")
        
        return False


class PushNotificationReceiverAuth(PushNotificationAuth):
    def __init__(self):
        self.public_keys_jwks = []
        self.jwks_client: PyJWKClient | None = None

    async def load_jwks(self, jwks_url: str):
        self.jwks_client = PyJWKClient(jwks_url)

    async def verify_push_notification(self, request: Request) -> bool:
        """
        [방어 강화 - 최종 동결] 
        - jwks_client 미 초기화 시 명시적 예외 방어
        - JWT/JWKS 검증 실패 시 예외 폭발 대신 안전한 False 반환 및 보안 감사 로그(Audit Log) 기록
        - 양방향 iat 검증을 통해 미래 시간 토큰 및 재생(Replay) 공격 완전 차단
        """
        if not self.jwks_client:
            logger.error("Security Audit: JWKS client is not initialized. Call load_jwks() first.")
            raise RuntimeError("JWKS client is not initialized")

        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith(AUTH_HEADER_PREFIX):
            logger.warning("Security Audit: Missing or malformed authorization header.")
            return False

        try:
            token = auth_header[len(AUTH_HEADER_PREFIX) :]
            signing_key = self.jwks_client.get_signing_key_from_jwt(token)

            decode_token = jwt.decode(
                token,
                signing_key.key,
                options={"require": ["iat", "request_body_sha256"]},
                algorithms=["RS256"],
            )

            # [방어 강화 - 양방향 iat 시간창 검증] 미래 시간 토큰 및 만료된 토큰 동시 차단
            now = time.time()
            iat = decode_token.get("iat", 0)
            token_age = now - iat

            if token_age < 0:
                logger.error("Security Audit: Rejected token with a future iat timestamp (Potential time-skew/spoofing attack).")
                return False
            
            if token_age > MAX_TOKEN_AGE_SECONDS:
                logger.warning("Security Audit: Rejected expired token (Exceeded 5-minute replay window).")
                return False

            # 요청 바디 SHA-256 무결성 검증
            body_json = await request.json()
            actual_body_sha256 = self._calculate_request_body_sha256(body_json)
            
            if actual_body_sha256 != decode_token["request_body_sha256"]:
                logger.error("Security Audit: Payload signature mismatch (Potential request body tampering).")
                return False

            return True

        except PyJWTError as e:
            logger.warning(f"Security Audit: JWT validation failed due to cryptographic/format error: {type(e).__name__}")
            return False
        except Exception as e:
            logger.error(f"Security Audit: Unexpected error during push notification verification: {type(e).__name__}")
            return False

최종 개선사항
✅ 광범위한 인증·네트워크 예외 처리 → HTTPStatusError·RequestError·PyJWTError 등 원인별 분리 → 장애 원인 추적성과 인증 실패 격리 강화
✅ jwks_client·private key 초기화 상태 의존 → 명시적 lifecycle 검증 → 초기화 누락에 따른 런타임 장애 방지
✅ 단방향 iat 만료 검사 → 미래 시각과 5분 초과 토큰을 모두 차단하는 시간창 검증 → Replay 및 시간 조작 방어 강화
✅ JWT 서명 검증 후 payload 무결성만 확인 → JWT·request_body_sha256·실제 요청 body를 순차 검증 → 인증 및 데이터 무결성 경계 강화
✅ 인증 실패를 단순 예외/출력으로 처리 → 보안 감사 로그와 안전한 False 반환으로 통일 → 외부 공격 시도 추적성과 서비스 안정성 확보
✅ Push URL·요청 payload 원문 로깅 → 상태·예외 유형 중심의 최소 로그로 전환 → 민감정보 및 로그 오염 위험 감소
✅ private key 미생성 상태 방치 → _generate_jwt()에서 명시적 검증 → 송신 인증 계층의 잘못된 초기화 조기 차단

원본의 RSA/JWKS·JWT·SHA-256 설계는 유지하면서 인증 실패·시간 검증·키 lifecycle·민감정보 노출 경계를 잠가, 과설계 없이 실무 보안 모듈 수준으로 승격한 버전이다.
