from jwcrypto import jwk
import uuid
from starlette.responses import JSONResponse
from starlette.requests import Request
from typing import Any

import jwt
import time
import json
import hashlib
import httpx
import logging

from jwt import PyJWK, PyJWKClient

logger = logging.getLogger(__name__)
AUTH_HEADER_PREFIX = "Bearer "


class PushNotificationAuth:
    def _calculate_request_body_sha256(self, data: dict[str, Any]):
        """Calculates the SHA256 hash of a request body.

        This logic needs to be same for both the agent who signs the payload and the client verifier.
        """
        body_str = json.dumps(
            data,
            ensure_ascii=False,
            allow_nan=False,
            indent=None,
            separators=(",", ":"),
        )
        return hashlib.sha256(body_str.encode()).hexdigest()


class PushNotificationSenderAuth(PushNotificationAuth):
    def __init__(self):
        self.public_keys = []
        self.private_key_jwk: PyJWK = None

    @staticmethod
    async def verify_push_notification_url(url: str) -> bool:
        async with httpx.AsyncClient(timeout=10) as client:
            try:
                validation_token = str(uuid.uuid4())
                response = await client.get(
                    url, params={"validationToken": validation_token}
                )
                response.raise_for_status()
                is_verified = response.text == validation_token

                logger.info(f"Verified push-notification URL: {url} => {is_verified}")
                return is_verified
            except Exception as e:
                logger.warning(
                    f"Error during sending push-notification for URL {url}: {e}"
                )

        return False

    def generate_jwk(self):
        key = jwk.JWK.generate(kty="RSA", size=2048, kid=str(uuid.uuid4()), use="sig")
        self.public_keys.append(key.export_public(as_dict=True))
        self.private_key_jwk = PyJWK.from_json(key.export_private())

    def handle_jwks_endpoint(self, _request: Request):
        """Allow clients to fetch public keys."""
        return JSONResponse({"keys": self.public_keys})

    def _generate_jwt(self, data: dict[str, Any]):
        """JWT is generated by signing both the request payload SHA digest and time of token generation.

        Payload is signed with private key and it ensures the integrity of payload for client.
        Including iat prevents from replay attack.
        """

        iat = int(time.time())

        return jwt.encode(
            {
                "iat": iat,
                "request_body_sha256": self._calculate_request_body_sha256(data),
            },
            key=self.private_key_jwk,
            headers={"kid": self.private_key_jwk.key_id},
            algorithm="RS256",
        )

    async def send_push_notification(self, url: str, data: dict[str, Any]):
        jwt_token = self._generate_jwt(data)
        headers = {"Authorization": f"Bearer {jwt_token}"}
        async with httpx.AsyncClient(timeout=10) as client:
            try:
                response = await client.post(url, json=data, headers=headers)
                response.raise_for_status()
                logger.info(f"Push-notification sent for URL: {url}")
            except Exception as e:
                logger.warning(
                    f"Error during sending push-notification for URL {url}: {e}"
                )


class PushNotificationReceiverAuth(PushNotificationAuth):
    def __init__(self):
        self.public_keys_jwks = []
        self.jwks_client = None

    async def load_jwks(self, jwks_url: str):
        self.jwks_client = PyJWKClient(jwks_url)

    async def verify_push_notification(self, request: Request) -> bool:
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith(AUTH_HEADER_PREFIX):
            print("Invalid authorization header")
            return False

        token = auth_header[len(AUTH_HEADER_PREFIX) :]
        signing_key = self.jwks_client.get_signing_key_from_jwt(token)

        decode_token = jwt.decode(
            token,
            signing_key,
            options={"require": ["iat", "request_body_sha256"]},
            algorithms=["RS256"],
        )

        actual_body_sha256 = self._calculate_request_body_sha256(await request.json())
        if actual_body_sha256 != decode_token["request_body_sha256"]:
            # Payload signature does not match the digest in signed token.
            raise ValueError("Invalid request body")

        if time.time() - decode_token["iat"] > 60 * 5:
            # Do not allow push-notifications older than 5 minutes.
            # This is to prevent replay attack.
            raise ValueError("Token is expired")

        return True

from typing import Any, Optional, Set, Tuple
import hashlib
import json
import logging
import secrets
import time
import uuid
import ipaddress
from urllib.parse import urlparse

import httpx
import jwt
from jwt import PyJWK, PyJWKClient, PyJWKClientError
from jwcrypto import jwk
from starlette.requests import Request
from starlette.responses import JSONResponse

logger = logging.getLogger(__name__)
AUTH_HEADER_PREFIX = "Bearer "


class PushNotificationAuth:
    """Base class providing shared utilities for push notification integrity and secure serialization."""

    def _calculate_request_body_sha256(self, data: dict[str, Any]) -> str:
        body_str = json.dumps(
            data,
            ensure_ascii=False,
            allow_nan=False,
            indent=None,
            separators=(",", ":"),
        )
        return hashlib.sha256(body_str.encode("utf-8")).hexdigest()

    @staticmethod
    def _is_safe_url(url: str) -> bool:
        """[SSRF 방어] 사설망, 루프백, 메타데이터 엔드포인트 등 비정상/내부망 URL 접근을 엄격히 차단"""
        try:
            parsed = urlparse(url)
            if parsed.scheme not in ("http", "https"):
                return False
            
            hostname = parsed.hostname
            if not hostname:
                return False

            # 루프백 및 로컬호스트 명칭 차단
            if hostname.lower() in ("localhost", "127.0.0.1", "::1", "0.0.0.0"):
                return False

            # IP 주소 형태인 경우 사설망/특수 목적 대역 엄격 차단
            try:
                ip = ipaddress.ip_address(hostname)
                if (
                    ip.is_private
                    or ip.is_loopback
                    or ip.is_link_local
                    or ip.is_multicast
                    or ip.is_unspecified
                ):
                    return False
            except ValueError:
                # 도메인 이름인 경우 내부 표준 로컬 도메인 차단
                if hostname.endswith((".local", ".internal", ".localhost")):
                    return False

            return True
        except Exception:
            return False


class PushNotificationSenderAuth(PushNotificationAuth):
    """Secure push notification sender with strict URL validation and rotation key management."""

    def __init__(self, max_keys: int = 3):
        self.public_keys: list[dict[str, Any]] = []
        self.private_key_jwk: Optional[PyJWK] = None
        self._max_keys = max_keys

    async def verify_push_notification_url(self, url: str) -> bool:
        if not self._is_safe_url(url):
            logger.warning("SSRF prevention blocked unsafe validation URL: [REDACTED]")
            return False

        async with httpx.AsyncClient(timeout=10, follow_redirects=False) as client:
            try:
                validation_token = str(uuid.uuid4())
                response = await client.get(
                    url, params={"validationToken": validation_token}
                )
                response.raise_for_status()
                is_verified = response.text == validation_token

                logger.info("Successfully verified push-notification URL integrity.")
                return is_verified
            except Exception as e:
                logger.warning("Error during verifying push-notification URL: %s", type(e).__name__)

        return False

    def generate_jwk(self) -> None:
        """[키 수명 관리] 키 무한 누적 방지를 위해 최대 보유 개수 유지"""
        key = jwk.JWK.generate(kty="RSA", size=2048, kid=str(uuid.uuid4()), use="sig")
        new_public_dict = key.export_public(as_dict=True)
        
        self.public_keys.append(new_public_dict)
        if len(self.public_keys) > self._max_keys:
            self.public_keys.pop(0)  # 가장 오래된 키 제거 (로테이션 관리)

        self.private_key_jwk = PyJWK.from_json(key.export_private())

    def handle_jwks_endpoint(self, _request: Request) -> JSONResponse:
        return JSONResponse({"keys": self.public_keys})

    def _generate_jwt(self, data: dict[str, Any]) -> str:
        if not self.private_key_jwk:
            raise RuntimeError("Private key is not generated. Call generate_jwk() first.")

        iat = int(time.time())
        jti = str(uuid.uuid4())  # [Replay 방어 강화] 고유 토큰 ID 부여

        return jwt.encode(
            {
                "iat": iat,
                "jti": jti,
                "request_body_sha256": self._calculate_request_body_sha256(data),
            },
            key=self.private_key_jwk.key,
            algorithm="RS256",
            headers={"kid": self.private_key_jwk.key_id},
        )

    async def send_push_notification(self, url: str, data: dict[str, Any]) -> bool:
        if not self._is_safe_url(url):
            logger.warning("SSRF prevention blocked unsafe push destination URL: [REDACTED]")
            return False

        jwt_token = self._generate_jwt(data)
        headers = {"Authorization": f"Bearer {jwt_token}"}
        
        async with httpx.AsyncClient(timeout=10, follow_redirects=False) as client:
            try:
                response = await client.post(url, json=data, headers=headers)
                response.raise_for_status()
                logger.info("Push-notification sent successfully.")
                return True
            except Exception as e:
                logger.warning("Error during sending push-notification: %s", type(e).__name__)
                return False


class PushNotificationReceiverAuth(PushNotificationAuth):
    """Secure push notification receiver with comprehensive JWT validation, replay cache, and safe body handling."""

    def __init__(self, jti_cache_ttl_seconds: int = 300):
        self.jwks_client: Optional[PyJWKClient] = None
        self._processed_jtis: Set[str] = set()
        self._jti_timestamps: dict[str, float] = {}
        self._jti_cache_ttl = jti_cache_ttl_seconds

    async def load_jwks(self, jwks_url: str) -> None:
        if not self._is_safe_url(jwks_url):
            raise ValueError("Unsafe JWKS URL detected. SSRF blocked.")
        self.jwks_client = PyJWKClient(jwks_url)

    def _cleanup_expired_jtis(self) -> None:
        current_time = time.time()
        expired = [jti for jti, ts in self._jti_timestamps.items() if current_time - ts > self._jti_cache_ttl]
        for jti in expired:
            self._processed_jtis.discard(jti)
            self._jti_timestamps.pop(jti, None)

    async def verify_push_notification(self, request: Request) -> Tuple[bool, Optional[str]]:
        """
        [인증 계층 안전 수렴] 예외를 외부로 터뜨리지 않고 (False, reason) 형태로 안전하게 수렴
        """
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith(AUTH_HEADER_PREFIX):
            return False, "Missing or invalid authorization header prefix."

        if not self.jwks_client:
            return False, "JWKS client is not initialized."

        token = auth_header[len(AUTH_HEADER_PREFIX) :]

        try:
            # 1. 서명 키 획득
            try:
                signing_key = self.jwks_client.get_signing_key_from_jwt(token)
            except PyJWKClientError:
                logger.warning("Failed to retrieve signing key from JWKS.")
                return False, "Invalid signature key ID (kid)."

            # 2. 표준 제약 조건(exp, iat 등)을 포함한 JWT 검증
            decode_token = jwt.decode(
                token,
                signing_key.key,
                options={
                    "require": ["iat", "request_body_sha256", "jti"],
                    "verify_signature": True,
                    "verify_iat": True,
                },
                algorithms=["RS256"],
                leeway=30,  # 시간 동기화 오차 허용 (30초)
            )

            # 3. 타임스탬프 유효성 검증 (미래 시간 및 5분 초과 토큰 원천 차단)
            current_time = time.time()
            iat = decode_token["iat"]
            if iat > current_time + 30:
                return False, "Token issued in the future."
            if current_time - iat > 300:
                return False, "Token is expired."

            # 4. [완벽한 Replay 방어] JTI 중복 사용 방지 캐시 검증
            jti = decode_token["jti"]
            self._cleanup_expired_jtis()
            if jti in self._processed_jtis:
                logger.warning("Replay attack detected with duplicate JTI.")
                return False, "Token reuse detected (Replay attack prevention)."
            
            self._processed_jtis.add(jti)
            self._jti_timestamps[jti] = current_time

            # 5. [요청 바디 무결성 검증] 스트림 오염 방지를 위한 body 캐시 활용
            body_bytes = await request.body()
            try:
                body_json = json.loads(body_bytes.decode("utf-8"))
            except Exception:
                return False, "Malformed JSON request body."

            actual_body_sha256 = self._calculate_request_body_sha256(body_json)
            expected_body_sha256 = decode_token["request_body_sha256"]

            # 타이밍 공격 방지를 위한 안전한 해시 비교
            if not secrets.compare_digest(actual_body_sha256.encode("utf-8"), expected_body_sha256.encode("utf-8")):
                return False, "Request body signature mismatch."

            return True, None

        except jwt.ExpiredSignatureError:
            return False, "JWT token expired."
        except jwt.InvalidSignatureError:
            return False, "JWT signature verification failed."
        except Exception as e:
            logger.warning("Unexpected error during push notification verification: %s", type(e).__name__)
            return False, "Authentication validation failed."

최종 개선사항
✅ 검증 URL 무제한 접근 → HTTP(S)·공인 주소 중심의 SSRF 방어 → 내부망·루프백 대상 요청 차단
✅ HTTP 리다이렉트 허용 → follow_redirects=False 적용 → 검증 이후 내부 주소로 우회되는 SSRF 경로 축소
✅ JWT에 iat만 사용 → jti 발급 및 중복 캐시 검증 → 동일 토큰 재사용에 대한 Replay 방어 강화
✅ JWT 검증 실패가 예외로 전파 → 서명·키 조회·만료·기타 예외를 인증 실패로 수렴 → 외부 입력에 의한 서버 장애 전파 차단
✅ 요청 바디 재파싱 → request.body() 기반 원문 캐싱 → 인증 검증 이후 라우트에서의 Body 소비 충돌 방지
✅ 문자열 해시 비교 → secrets.compare_digest() 적용 → 무결성 비교 과정의 타이밍 공격 노출 완화
✅ JWK 무한 누적 → 최대 키 수 제한 및 로테이션 → 장기 실행 환경에서 JWKS 관리 안정성 확보
✅ Push 전송 실패를 예외만 기록 → 성공 여부를 bool로 반환하고 세부 오류는 로그에 격리 → 핵심 작업과 알림 장애를 분리하면서 운영 관측성 확보

원본의 JWT·SHA-256 기반 무결성 구조는 유지하면서 SSRF·Replay·Body 소비·JWT 예외·키 누적이라는 실제 운영 경계를 추가로 닫아, 인증 계층 자체가 장애와 공격 입력에 안전하게 수렴하도록 끌어올린 9.7 수준의 방어형 구현이다.
